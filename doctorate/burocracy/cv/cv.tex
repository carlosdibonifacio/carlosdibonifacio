% Entry with institution, location, degree/role, and dates
% #1 = Institution, #2 = Location, #3 = Degree/Role, #4 = Dates
\newcommand{\cventry}[4]{%
  \noindent
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{#1} & \textcolor{lightgray}{\small #2} \\
    \textit{#3} & \textcolor{lightgray}{\small #4} \\
  \end{tabular*}%
  \vspace{3pt}%
}

% Entry with one line (e.g., awards)
% #1 = Title/Award, #2 = Dates
\newcommand{\cvaward}[2]{%
  \noindent
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    #1 & \textcolor{lightgray}{\small #2} \\
  \end{tabular*}%
  \vspace{2pt}%
}

% Conference/workshop entry
% #1 = Event name, #2 = Location, #3 = Role (presenter/attendee/etc.), #4 = Date

  \noindent

    \textbf{#1} & \textcolor{lightgray}{\small #4} \\
    \textit{#3} \textcolor{lightgray}{$\cdot$ #2} & \\
  \end{tabular*}%
  \vspace{3pt}%
}
